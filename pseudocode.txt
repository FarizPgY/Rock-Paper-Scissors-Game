FUNCTION getComputerChoice()
    SET randomNumber to random number between 1 and 3
    IF randomNumber is equal to 1 THEN
        SET choice to "rock"
    IF ELSE randomNumber is equal to 2 THEN
        SET choice to "paper"
    ELSE 
        SET choice to "scissors"
    RETURN choice
ENDFUCTION

FUNCTION getHumanChoice()
    INPUT choice (rock/paper/scissors)
    RETURN choice
ENDFUCTION

FUNCTION playRound(humanChoice, computerChoice)
    SET humanChoice to lowercase
    IF humanChoice is equal to computerChoice THEN
        DISPLAY "We Draw!"
    ELSE

        IF humanChoice is "rock" and computerChoice is "scissors" THEN
            DISPLAY "You win! rock beats scissors"
            SET humanScore plus one

        IF ELSE humanChoice is "rock" and computerChoice is "paper" THEN
            DISPLAY "You lose! paper beats rock"
            SET computerScore plus one

        IF ELSE humanChoice is "paper" and computerChoice is "rock" THEN
            DISPLAY "You win! paper beats rock"
            SET humanScore plus one

        IF ELSE humanChoice is "paper" and computerChoice is "scissors" THEN
            DISPLAY "You lose! scissors beats paper"
            SET computerScore plus one

        IF ELSE humanChoice is "scissors" and computerChoice is "paper" THEN
            DISPLAY "You win! scissors beats paper"
            SET humanScore plus one

        ELSE 
            DISPLAY "You lose! rock beats scissors"
            SET computerScore plus one
        
ENDFUCTION

SET computerScore to 0
SET humanScore to 0
